0. 公共环境与 AqSolDB 数据准备

In [2] · Python
import os
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from rdkit import Chem, DataStructs, RDLogger
from rdkit.Chem import Descriptors, Draw, rdFingerprintGenerator

from sklearn.model_selection import train_test_split, KFold, GridSearchCV
from sklearn.preprocessing import StandardScaler, PolynomialFeatures
from sklearn.pipeline import make_pipeline
from sklearn.linear_model import LinearRegression, Ridge, Lasso
from sklearn.neighbors import KNeighborsRegressor
from sklearn.svm import SVR, SVC
from sklearn.ensemble import RandomForestRegressor, RandomForestClassifier
from sklearn.metrics import (
    root_mean_squared_error, r2_score,
    accuracy_score, precision_score, recall_score, f1_score, roc_auc_score,
)
from xgboost import XGBRegressor, XGBClassifier

RDLogger.DisableLog("rdApp.*")

for p in ["../data/solubility.csv", "data/solubility.csv", "../../01 - Basics/data/solubility.csv"]:
    if os.path.exists(p):
        df = pd.read_csv(p)
        break

df["mol"] = df["SMILES"].apply(Chem.MolFromSmiles)
df = df[df["mol"].notna()].reset_index(drop=True)

feature_cols = [
    "MolWt", "MolLogP", "MolMR", "HeavyAtomCount", "NumHAcceptors", "NumHDonors",
    "NumHeteroatoms", "NumRotatableBonds", "NumValenceElectrons", "NumAromaticRings",
    "NumSaturatedRings", "NumAliphaticRings", "RingCount", "TPSA", "LabuteASA",
    "BalabanJ", "BertzCT",
]
X = df[feature_cols].values
y = df["Solubility"].values
idx_train, idx_test = train_test_split(np.arange(len(df)), test_size=0.2, random_state=42)
X_train, X_test = X[idx_train], X[idx_test]
y_train, y_test = y[idx_train], y[idx_test]

scaler = StandardScaler().fit(X_train)
X_train_s, X_test_s = scaler.transform(X_train), scaler.transform(X_test)
print(f"Prepared {len(df)} molecules.")
Out:
Prepared 9480 molecules.

    ("Underfit (C=0.1, gamma=0.005)", dict(C=0.1, gamma=0.005)),

02e 模型验证、调参与适用域练习答案

练习 02e-1：防泄漏 Pipeline + GridSearchCV 调优 kNN

In [12] · Python
knn_pipe = make_pipeline(StandardScaler(), KNeighborsRegressor())
cv5 = KFold(n_splits=5, shuffle=True, random_state=42)
grid_knn = GridSearchCV(
    knn_pipe,
    param_grid={
        "kneighborsregressor__n_neighbors": [3, 5, 10, 20],
        "kneighborsregressor__weights": ["uniform", "distance"],
    },
    cv=cv5,
    scoring="neg_root_mean_squared_error",
    n_jobs=-1,
).fit(X_train, y_train)

print("最优超参数组合:", grid_knn.best_params_)
print(f"最优 5-Fold CV RMSE = {-grid_knn.best_score_:.3f}")
print(f"独立测试集 RMSE     = {root_mean_squared_error(y_test, grid_knn.best_estimator_.predict(X_test)):.3f}")
Out:

train_bit_fps = [fp_gen.GetFingerprint(df["mol"][i]) for i in idx_train]
train_cnt_fps = [fp_gen.GetCountFingerprint(df["mol"][i]) for i in idx_train]

osc_candidates = {
    "Pentacene": "c1ccc2cc3cc4cc5ccccc5cc4cc3cc2c1",
    "3-Hexylthiophene": "CCCCCCc1ccsc1",
    "BTBT": "c1ccc2c(c1)sc1c2sc2ccccc21",
    "C8-BTBT": "CCCCCCCCc1ccc2c(c1)sc1c3ccc(CCCCCCCC)cc3sc21",
    "Rubrene": "c1ccc(cc1)-c1c2ccccc2c(-c2ccccc2)c2c(-c3ccccc3)c3ccccc3c(-c3ccccc3)c12",
    "TIPS-Pentacene": "CC(C)[Si](C#Cc1c2cc3ccccc3cc2c(C#C[Si](C(C)C)(C(C)C)C(C)C)c2cc3ccccc3cc12)(C(C)C)C(C)C",
    "EDOT": "C1COc2cscc2O1",
}

rows_cmp = []
for name, smi in osc_candidates.items():
    mol = Chem.MolFromSmiles(smi)
    sim_bit = max(DataStructs.BulkTanimotoSimilarity(fp_gen.GetFingerprint(mol), train_bit_fps))
    sim_cnt = max(DataStructs.BulkTanimotoSimilarity(fp_gen.GetCountFingerprint(mol), train_cnt_fps))
    rows_cmp.append({"Candidate": name, "Max_BitFP_Sim": round(sim_bit, 3), "Max_CountFP_Sim": round(sim_cnt, 3)})

print(pd.DataFrame(rows_cmp).to_string(index=False))
